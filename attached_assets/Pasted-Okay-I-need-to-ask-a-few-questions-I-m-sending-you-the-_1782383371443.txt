Android Setup

Configure Dependencies



1. Update your Project-Level build.gradle with the following repositories:



allprojects {

  repositories {

    google()

    jcenter()

    mavenCentral()

    maven { url "https://artifact.bytedance.com/repository/pangle" }

    maven { url "https://android-sdk.is.com" }

    maven { url "https://dl-maven-android.mintegral.com/repository/mbridge_android_sdk_oversea" }

    maven { url "https://artifactory.bidmachine.io/bidmachine" }

    maven { url "https://ysonetwork.s3.eu-west-3.amazonaws.com/sdk/android" }

    maven { url "https://repo.pubmatic.com/artifactory/public-repos" }

    maven { url "https://bitbucket.org/sdkcenter/sdkcenter/raw/release" }

  }

}



2. Add dependencies to your app-level build.gradle:



implementation("com.yodo1.mas:full:4.15.0")

implementation("com.google.android.gms:play-services-ads:23.1.0")

implementation("com.google.ads.mediation:pangle:6.0.0.3.0")

implementation("io.bidmachine:ads:2.7.0")



Configure Project Settings

1. Enable multiDex in build.gradle:



defaultConfig {

  multiDexEnabled true

  minSdkVersion 24

}



2. Configure AndroidX and Jetifier in gradle.properties:



android.useAndroidX=true

android.enableJetifier=true

android.enableDexingArtifactTransform=false



3. Update AndroidManifest.xml:



<manifest xmlns:tools="http://schemas.android.com/tools">

    <application

        tools:replace="android:label"

        tools:targetApi="n">

        <!-- Sample AdMob ID: ca-app-pub-3940256099942544~3347511713 -->

        <meta-data

            android:name="com.google.android.gms.ads.APPLICATION_ID"

            android:value="YOUR_ADMOB_APP_ID"/>

    </application>

</manifest>
